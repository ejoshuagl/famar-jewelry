    68→

    70→  const { adminName } = useAuthStore()
    71→  const queryClient = useQueryClient()

    73→  const [search, setSearch] = useState('')

    75→  const [detailDialogOpen, setDetailDialogOpen] = useState(false)
    76→  const [editDialogOpen, setEditDialogOpen] = useState(false)

    78→  const [selectedOrder, setSelectedOrder] = useState<Record<string, unknown> | null>(null)
    79→
    80→  // Edit state
    81→  const [editItems, setEditItems] = useState<EditableOrderItem[]>([])
    82→  const [editObs, setEditObs] = useState('')
    83→  const [editName, setEditName] = useState('')
    84→  const [editCity, setEditCity] = useState('')
    85→  const [editPhone, setEditPhone] = useState('')
    86→
    87→  const { data, isLoading } = useQuery({
    88→    queryKey: ['admin-orders', statusFilter, search, page],
    89→    queryFn: async () => {
    90→      const params = new URLSearchParams({
    91→        limit: '20',
    92→        page: page.toString(),
    93→      })
    94→      if (statusFilter !== 'all') params.set('status', statusFilter)
    95→      if (search) params.set('search', search)
    96→      const res = await fetch(`/api/orders?${params}`, {
    97→        headers: { 'x-admin-name': adminName || '' },
    98→      })
    99→      if (!res.ok) throw new Error('Unauthorized')
   100→      return res.json()
   101→    },
   102→  })
   103→
   104→  const confirmMutation = useMutation({
   105→    mutationFn: async (orderId: string) => {
   106→      const res = await fetch(`/api/orders/${orderId}`, {
   107→        method: 'PUT',
   108→        headers: {
   109→          'Content-Type': 'application/json',
   110→          'x-admin-name': adminName || '',
   111→        },
   112→        body: JSON.stringify({ status: 'confirmed' }),
   113→      })
   114→      if (!res.ok) {
   115→        const d = await res.json()
   116→        throw new Error(d.error || 'Error confirming order')
   117→      }
   118→      return res.json()
   119→    },
   120→    onSuccess: () => {
   121→      queryClient.invalidateQueries({ queryKey: ['admin-orders'] })
   122→      queryClient.invalidateQueries({ queryKey: ['admin-stats'] })
   123→      toast.success('Pedido confirmado. Stock actualizado.')
   124→      setDetailDialogOpen(false)
   125→    },
   126→    onError: (err) => toast.error(err.message || 'Error al confirmar el pedido'),
   127→  })
   128→
   129→  const cancelMutation = useMutation({
   130→    mutationFn: async (orderId: string) => {
   131→      const res = await fetch(`/api/orders/${orderId}`, {
   132→        method: 'PUT',
   133→        headers: {
   134→          'Content-Type': 'application/json',
   135→          'x-admin-name': adminName || '',
   136→        },
   137→        body: JSON.stringify({ status: 'cancelled' }),
   138→      })
   139→      if (!res.ok) {
   140→        const d = await res.json()
   141→        throw new Error(d.error || 'Error cancelling order')
   142→      }
   143→      return res.json()
   144→    },
   145→    onSuccess: () => {
   146→      queryClient.invalidateQueries({ queryKey: ['admin-orders'] })
   147→      queryClient.invalidateQueries({ queryKey: ['admin-stats'] })

   149→      setDetailDialogOpen(false)
   150→    },
   151→    onError: (err) => toast.error(err.message || 'Error al cancelar el pedido'),
   152→  })

   155→    mutationFn: async (orderId: string) => {
   156→      const res = await fetch(`/api/orders/${orderId}`, {
   157→        method: 'DELETE',
   158→        headers: { 'x-admin-name': adminName || '' },
   159→      })
   160→      if (!res.ok) {
   161→        const d = await res.json()
   162→        throw new Error(d.error || 'Error deleting order')
   163→      }
   164→      return res.json()
   165→    },
   166→    onSuccess: () => {
   167→      queryClient.invalidateQueries({ queryKey: ['admin-orders'] })
   168→      queryClient.invalidateQueries({ queryKey: ['admin-stats'] })
   169→      toast.success('Pedido eliminado permanentemente')
   170→      setDetailDialogOpen(false)
   171→      setDeleteDialogOpen(false)
   172→      setSelectedOrder(null)
   173→    },
   174→    onError: (err) => toast.error(err.message || 'Error al eliminar el pedido'),
   175→  })

   178→    mutationFn: async () => {
   179→      if (!selectedOrder) return

   181→      const res = await fetch(`/api/orders/${selectedOrder.id}`, {
   182→        method: 'PUT',
   183→        headers: {
   184→          'Content-Type': 'application/json',
   185→          'x-admin-name': adminName || '',
   186→        },
   187→        body: JSON.stringify({
   188→          items: editItems,
   189→          total,
   190→          observations: editObs,
   191→          customerName: editName,
   192→          customerCity: editCity,
   193→          customerPhone: editPhone,
   194→        }),
   195→      })
   196→      if (!res.ok) {
   197→        const d = await res.json()
   198→        throw new Error(d.error || 'Error saving order')
   199→      }
   200→      return res.json()
   201→    },
   202→    onSuccess: () => {
   203→      queryClient.invalidateQueries({ queryKey: ['admin-orders'] })
   204→      queryClient.invalidateQueries({ queryKey: ['admin-stats'] })
   205→      toast.success('Pedido modificado exitosamente')
   206→      setEditDialogOpen(false)
   207→      setDetailDialogOpen(false)
   208→    },
   209→    onError: (err) => toast.error(err.message || 'Error al guardar cambios'),
   210→  })

   213→  const totalPages = data?.totalPages || 1
   214→

   217→      pending: 'bg-amber-100 text-amber-700 dark:bg-amber-900 dark:text-amber-300',

   219→      cancelled: 'bg-red-100 text-red-700 dark:bg-red-900 dark:text-red-300',
   220→    }

   222→      pending: 'Pendiente',
   223→      confirmed: 'Confirmado',
   224→      cancelled: 'Cancelado',
   225→    }

   227→      <span className={`text-xs px-2 py-0.5 rounded-full font-medium ${styles[status] || ''}`}>
   228→        {labels[status] || status}
   229→      </span>
   230→    )
   231→  }
   232→
   233→  const openDetail = (order: Record<string, unknown>) => {

   235→    setDetailDialogOpen(true)
   236→  }
   237→

   243→      id: item.id as string,
   244→      productId: item.productId as string,
   245→      quantity: item.quantity as number,
   246→      price: item.price as number,
   247→      name: item.name as string,
   248→      code: item.code as string,
   249→    }))

   251→    setEditObs((target.observations as string) || '')

   253→    setEditCity(target.customerCity as string)

   255→    setDetailDialogOpen(false)
   256→    setEditDialogOpen(true)
   257→  }
   258→
   259→  const handleItemQtyChange = (itemId: string, delta: number) => {
   260→    setEditItems((prev) =>
   261→      prev.map((item) =>
   262→        item.id === itemId
   263→          ? { ...item, quantity: Math.max(1, item.quantity + delta) }
   264→          : item
   265→      )
   266→    )
   267→  }
   268→
   269→  const handleRemoveItem = (itemId: string) => {
   270→    setEditItems((prev) => prev.filter((item) => item.id !== itemId))
   271→  }
   272→
   273→  const editTotal = editItems.reduce((sum, i) => sum + i.quantity * i.price, 0)

   279→        <div className="flex flex-col sm:flex-row gap-3">
   280→          <div className="relative flex-1 max-w-sm">
   281→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />

   283→              placeholder="Buscar por nombre, teléfono o #pedido..."
   284→              value={search}
   285→              onChange={(e) => { setSearch(e.target.value); setPage(1) }}

   289→          <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setPage(1) }}>
   290→            <SelectTrigger className="w-full sm:w-40">
   291→              <SelectValue placeholder="Estado" />
   292→            </SelectTrigger>
   293→            <SelectContent>
   294→              <SelectItem value="all">Todos</SelectItem>
   295→              <SelectItem value="pending">Pendientes</SelectItem>
   296→              <SelectItem value="confirmed">Confirmados</SelectItem>
   297→              <SelectItem value="cancelled">Cancelados</SelectItem>
   298→            </SelectContent>
   299→          </Select>
   300→        </div>
   301→
   302→        {/* Desktop Table */}
   303→        <Card className="hidden lg:block">
   304→          <CardContent className="p-0">
   305→            <Table>
   306→              <TableHeader>
   307→                <TableRow>
   308→                  <TableHead>Pedido</TableHead>
   309→                  <TableHead>Cliente</TableHead>
   310→                  <TableHead>Ciudad</TableHead>
   311→                  <TableHead>Teléfono</TableHead>
   312→                  <TableHead>Total</TableHead>
   313→                  <TableHead>Estado</TableHead>
   314→                  <TableHead>Fecha</TableHead>
   315→                  <TableHead className="text-right">Acciones</TableHead>
   316→                </TableRow>
   317→              </TableHeader>
   318→              <TableBody>
   319→                {isLoading ? (
   320→                  Array.from({ length: 5 }).map((_, i) => (
   321→                    <TableRow key={i}>
   322→                      {Array.from({ length: 8 }).map((_, j) => (
   323→                        <TableCell key={j}><div className="h-4 w-full bg-muted rounded animate-pulse" /></TableCell>
   324→                      ))}
   325→                    </TableRow>
   326→                  ))
   327→                ) : orders.length === 0 ? (
   328→                  <TableRow>
   329→                    <TableCell colSpan={8} className="text-center py-8 text-muted-foreground">
   330→                      No se encontraron pedidos
   331→                    </TableCell>
   332→                  </TableRow>
   333→                ) : (
   334→                  orders.map((order: Record<string, unknown>) => (
   335→                    <TableRow key={order.id}>
   336→                      <TableCell className="font-medium text-sm">#{order.orderNumber}</TableCell>
   337→                      <TableCell className="text-sm">{order.customerName}</TableCell>
   338→                      <TableCell className="text-sm">{order.customerCity}</TableCell>
   339→                      <TableCell className="text-sm">{order.customerPhone}</TableCell>
   340→                      <TableCell className="text-sm font-medium">{formatPrice(order.total as number)}</TableCell>
   341→                      <TableCell>{getStatusBadge(order.status as string)}</TableCell>
   342→                      <TableCell className="text-xs text-muted-foreground">
   343→                        {new Date(order.createdAt as string).toLocaleDateString('es-EC')}
   344→                      </TableCell>
   345→                      <TableCell className="text-right">
   346→                        <div className="flex justify-end gap-1">
   347→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openDetail(order)}>
   348→                            <Eye className="h-3.5 w-3.5" />
   349→                          </Button>
   350→                          {(order.status as string) === 'pending' && (
   351→                            <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openEdit(order)}>
   352→                              <Pencil className="h-3.5 w-3.5" />
   353→                            </Button>
   354→                          )}
   355→                        </div>
   356→                      </TableCell>
   357→                    </TableRow>
   358→                  ))
   359→                )}
   360→              </TableBody>
   361→            </Table>
   362→          </CardContent>
   363→        </Card>
   364→
   365→        {/* Mobile Cards */}
   366→        <div className="lg:hidden space-y-2">
   367→          {isLoading ? (
   368→            Array.from({ length: 5 }).map((_, i) => (
   369→              <Card key={i} className="p-3">
   370→                <div className="flex justify-between mb-2">
   371→                  <div className="h-4 w-24 bg-muted rounded animate-pulse" />
   372→                  <div className="h-5 w-16 bg-muted rounded-full animate-pulse" />
   373→                </div>
   374→                <div className="h-3 w-3/4 bg-muted rounded animate-pulse mb-1" />
   375→                <div className="flex justify-between mt-3">
   376→                  <div className="h-4 w-20 bg-muted rounded animate-pulse" />
   377→                  <div className="h-8 w-8 bg-muted rounded animate-pulse" />
   378→                </div>
   379→              </Card>
   380→            ))
   381→          ) : orders.length === 0 ? (
   382→            <div className="text-center py-8 text-muted-foreground">
   383→              No se encontraron pedidos
   384→            </div>
   385→          ) : (
   386→            orders.map((order: Record<string, unknown>) => (
   387→              <Card key={order.id} className="p-3">
   388→                <div className="flex items-start justify-between gap-2">
   389→                  <div className="min-w-0 flex-1">
   390→                    <p className="font-medium text-sm">#{order.orderNumber}</p>
   391→                    <p className="text-sm text-muted-foreground truncate">{order.customerName}</p>
   392→                    <p className="text-xs text-muted-foreground">{order.customerCity}</p>
   393→                  </div>
   394→                  {getStatusBadge(order.status as string)}
   395→                </div>
   396→                <div className="flex items-center justify-between mt-3">
   397→                  <div>
   398→                    <p className="text-base font-bold">{formatPrice(order.total as number)}</p>
   399→                    <p className="text-xs text-muted-foreground">
   400→                      {new Date(order.createdAt as string).toLocaleDateString('es-EC')}
   401→                    </p>
   402→                  </div>
   403→                  <div className="flex gap-1">
   404→                    {(order.status as string) === 'pending' && (
   405→                      <Button variant="outline" size="sm" onClick={() => openEdit(order)}>
   406→                        <Pencil className="h-3.5 w-3.5 mr-1" />
   407→                        Editar
   408→                      </Button>
   409→                    )}
   410→                    <Button variant="outline" size="sm" onClick={() => openDetail(order)}>
   411→                      <Eye className="h-3.5 w-3.5 mr-1" />
   412→                      Ver
   413→                    </Button>
   414→                  </div>
   415→                </div>
   416→              </Card>
   417→            ))
   418→          )}
   419→        </div>
   420→
   421→        {totalPages > 1 && (
   422→          <div className="flex justify-center gap-1">
   423→            {Array.from({ length: totalPages }).map((_, i) => (
   424→              <Button
   425→                key={i}
   426→                variant={page === i + 1 ? 'default' : 'outline'}
   427→                size="sm"
   428→                className="h-8 w-8 p-0"
   429→                onClick={() => setPage(i + 1)}
   430→              >
   431→                {i + 1}
   432→              </Button>
   433→            ))}
   434→          </div>
   435→        )}
   436→
   437→        {/* ========== Order Detail Dialog ========== */}
   438→        <Dialog open={detailDialogOpen} onOpenChange={setDetailDialogOpen}>
   439→          <DialogContent className="max-w-lg max-h-[85vh]">
   440→            <DialogHeader>
   441→              <DialogTitle>Pedido #{selectedOrder?.orderNumber}</DialogTitle>
   442→              <DialogDescription>
   443→                Detalles del pedido del cliente
   444→              </DialogDescription>
   445→            </DialogHeader>
   446→            <div className="max-h-[65vh] overflow-y-auto space-y-4">
   447→            {selectedOrder && (
   448→              <>
   449→                <div className="grid grid-cols-2 gap-3 text-sm">
   450→                  <div>

   452→                    <p className="font-medium">{selectedOrder.customerName}</p>
   453→                  </div>

   455→                    <span className="text-muted-foreground">Ciudad:</span>

   466→                  <div>
   467→                    <span className="text-muted-foreground">Estado:</span>
   468→                    <div className="mt-1">{getStatusBadge(selectedOrder.status as string)}</div>
   469→                  </div>
   470→                </div>
   471→
   472→                <div>
   473→                  <h4 className="text-sm font-semibold mb-2">Productos</h4>

   476→                      <table className="w-full text-sm">

   479→                            <th className="text-left px-3 py-2 whitespace-nowrap">Producto</th>

   504→                              {formatPrice(selectedOrder.total as number)}
   505→                            </td>
   506→                          </tr>
   507→                        </tfoot>
   508→                      </table>
   509→                    </div>
   510→                  </div>
   511→                </div>
   512→
   513→                {selectedOrder.observations && (
